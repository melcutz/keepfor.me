Add this to enable Claude to search, read, and save to your Keepfor.me library:

{
  "mcpServers": {
    "keepfor-me": {
      "command": "uvx",
      "args": [
        "keepfor-me-mcp",
        "--url", "{{ base_url }}",
        "--token", "YOUR_PAT_TOKEN_HERE"
      ]
    }
  }
}
